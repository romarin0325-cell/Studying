class Text(HTMLParser):
    def __init__(self):super().__init__();self.parts=[];self.skip=0
    def handle_starttag(self,t,a):
        if t in ('style','script'):self.skip+=1
    def handle_endtag(self,t):
        if t in ('style','script') and self.skip:self.skip-=1
    def handle_data(self,s):
        if not self.skip and s.strip():self.parts.append(s.strip())
def fetch(spec):
    id,language,kind,url=spec;result={'id':id,'language':language,'kind':kind,'url':url}
    try:
        req=urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0'})
        with urllib.request.urlopen(req,timeout=24) as r:
            raw=r.read(3000000);result.update(status=r.status,final_url=r.url,content_type=r.headers.get('Content-Type',''))
        text=raw.decode('utf-8','replace')
        if '<html' in text[:2000].lower() or '<!doctype' in text[:200].lower():
            p=Text();p.feed(text);text='\n'.join(p.parts)
        (ROOT/(id+'.txt')).write_text(text)
        result.update(bytes=len(raw),text_sha256=hashlib.sha256(text.encode()).hexdigest(),text_file=id+'.txt')
    except Exception as e:result.update(status=getattr(e,'code',None),error=str(e))
    return result
